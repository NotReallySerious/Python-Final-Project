FUNCTION redo_action(prompt)
    WHILE True DO
        prompt user for enter yes or no (choice)
        IF choice LIKE 'yes' or 'y' DO
            OUTPUT True
        ELSE IF choice LIKE 'no' or 'n' DO
            OUTPUT False
        ELSE
            DISPLAY "Invalid Value. please enter 'yes', or 'no'."
        ENDIF
    ENDWHILE
ENDFUNCTION

FUNCTION get_receipt_list():
    SET receipts = empty dictionary 
    SET current_id = 1
    OPEN receipt_db.txt FOR READING:
        READ EVERY LINE IN receipt_db.txt

        FOR EACH value IN lines:

            SET data = SPLIT LINE BY ':'
            SET customer_name = trim whitespace from data[1]

            DISPLAY current_id + '.' + customer_name

            INCREMENT current_id BY 1
        
        ENDFOR

    WHILE True DO
        SET choice = PROMPT USER TO ENTER ID TO VIEW THE RECEIPT DETAILS (ENTER 00 TO EXIT)

        IF choice EQUALS TO '00' DO
            BREAK LOOP
        ENDIF

        TRY:
            SET choice = CONVERT choice INTO INTEGER
        CATCH:
            DISPLAY 'invalid ID'
            CONTINUE the loop iteration
        ENDTRY
        
        IF choice EXISTS IN receipt dictionary DO
            GET customer_name FROM receipts USING choice AS KEY INDICATOR
            CONSTRUCT receipt_file PATH AS "accountant/{customer_name_lowercase_with_underscores}_receipt.txt"
        ELSE
            DISPLAY 'Invalid ID'
            READ AND DISPLAY ENTIRE CONTENT IN THE receipt_db.txt
        ENDIF

        TRY:
            OPEN receipt_file FOR READING AS file
                DISPLAY 'Receipt for {customer_name}'
                READ AND DISPLAY THE ENTIRE receipt_file CONTENT
            CLOSE FILE
        CATCH FileNotFoundError
            DISPLAY 'Error : receipt for {customer_name} cant be found'
        ENDTRY
        
        // Ask the accountant if they want to see another receipt or not
        IF redo_action('see another receipt') IS FALSE:
            EXIT FUNCTION
        ENDIF
    ENDWHILE
ENDFUNCTION

FUNCTION delete_receipt():
    SET receipts as empty dictionary
    set current_id = 1
    OPEN accountant/receipt_db.txt FOR READING AS rl
        READ ALL LINES FROM rl INTO LINES 

        FOR EACH line IN lines DO
            EXTRACT customer_name from line (split by ':', take the second part, trim whitespace)
            STORE customer_name in receipts dictionary with key current_id 
            DISPLAY "{current_id}. {customer_name}"

            INCREMENT current_id by 1
    CLOSE rl

    DISPLAY 'Enter the receipt ID to Delete (00 to exit)'
    WHILE True DO 
        PROMPT USER TO ENTER RECEIPT ID (choice)

        IF choice EQUALS TO '00' THEN
            EXIT loop
        ENDIF

        IF lines IS EMPTY THEN
            DISPLAY 'No Receipts to delete'
            RETURN FROM FUNCTION
        ENDIF

        TRY
            CONVERT choice INTO INTEGER
        CATCH conversion error
            DISPLAY 'Invalid ID'
            CONTINUE to the next iteration
        ENDTRY

        IF choice EXISTS IN receipts dictionary THEN 
            GET customer_name from receipts using choice as key
            CONSTRUCT receipt_file path as "accountant/{customer_name_lowercase_with_underscores}_receipt.txt"
        ELSE
            DISPLAY 'Invalid ID'
            CONTINUE iteration
        ENDIF

        IF receipt_file EXISTS THEN
            DELETE receipt_file 
            DISPLAY "{customer_name}'s receipt has been deleted"
        ELSE
            DISPLAY "receipt for {customer_name} not found"
        ENDIF
        
        // REMOVE ENTRY 
        DELETE lines[choice - 1]

        // UPDATE THE DATABASE FILE 
        OPEN FILE 'accountant/receipt_db.txt' for WRITING AS rl
            WRITE ALL LINES TO rl
        CLOSE FILE

        // ASK IF THE ACCOUNTANT WANTS TO DELETE RECEIPT AGAIN
        IF redo_action('delete another receipt') IS FALSE
            EXIT FUNCTION
    ENDWHILE
ENDFUNCTION

FUNCTION daily_summary():

    SET patient_count = 0
    SET med_sold as dictionary
    SET patients as empty set
    SET total = 0
    SET Date 
    SET VALID_DATE = False

    // VALIDATE THE DATE INPUTTED
    
    While VALID_DATE is False THEN 
        PROMPT USER FOR ENTER THE DATE IN FORMAT OF YYYY-MM-DD (Date)

        IF Date IS EMPTY THEN
            Date = GET today's Date
            DISPLAY 'Use today's Date {date}'
            SET VALID_DATE = True
        ELSE
            TRY
                VALIDATE THE DATE USING THE FORMAT YYYY-MM-DD
                SET VALID_DATE = True
            CATCH ValueError:
                DISPLAY 'Invalid date format! Please use YYYY-MM-DD (e.g., 2025-12-30)'
            ENDTRY
        ENDIF
    ENDWHILE


    OPEN accountant/patient_billing_record.txt FOR READING AS f
        SET lines = read all the lines from file

        FOR every line in lines THEN

            IF line is empty THEN
                CONTINUE iteration
            ENDIF
            
            SET part = split by ';'
            EXTRACT date from part (trim whitespace, take the first element in part, replace "'" to "")
            EXTRACT patient_name from part (trim whitespace, take the second element in part, replace "'" to "")
            EXTRACT item_name from part (trim whitespace, take the third element in part, replace "'" to "")
            EXTRACT quantity from part (trim whitespace, take the fourth element in part, replace "'" to "", change to integer)
            EXTRACT price from part (trim whitespace, take the fifth element in part, replace "'" to "", change to float)

            IF Date EQUALS TO date:
                ADD patient_name to patients set
                med_sold[item_name] = med_sold.get(item_name, 0) + quantity
                INCREMENT total by (quantity times price)
            ENDIF
        ENDFOR
        SET patient_count = length of patients
    CLOSE FILE
            
    IF patient_count EQUALS TO 0 THEN
        DISPLAY 'no transaction found for {date}'
        EXIT FUNCTION
    ENDIF
    
    SET summary_file_path = "accountant/{Date}_daily_summary.txt"

    OPEN summary_file_path FOR WRITING AS sum 
        WRITE 'Asia Pacific Hospital Healthplus Sales Summary' + new line to sum
        WRITE '================================' + new line to sum
        WRITE 'Date: {Date}' + new line to sum
        WRITE 'Total patients: {patient_count}' + new line to sum
        WRITE 'Items sold' + new line to sum
        FOR each item, and quantity in med_sold dictionary DO
            WRITE '{item}: {quantity} x {price}' + new line to sum
        ENDFOR
        WRITE '=================================' + new line to sum
        WRITE 'Total Income = RM. {total (round up to 2 floating points)}' + new line to sum
    CLOSE FILE

    DISPLAY 'Daily Summary for today is saved on {summary_file_path}'

    IF user NOT redo_action('generate another summary') THEN
        EXIT FUNCTION
    ENDIF
ENDFUNCTION
